type ErrorResponseData = {
  message?: string
  errors?: Array<string | { code?: string; description?: string }>
}

export const getErrorMessage = (
  errors: ApiResponse<unknown>['errors'] | undefined,
  fallback: string,
) => {
  const firstError = errors?.[0]

  if (typeof firstError === 'string') {
    return firstError
  }

  return firstError?.description || firstError?.code || fallback
}

export const getHttpErrorMessage = (error: unknown, fallback: string) => {
  if (typeof error !== 'object' || error === null) return fallback

  const response = (error as { response?: { data?: ErrorResponseData } }).response
  return getErrorMessage(response?.data?.errors, response?.data?.message || fallback)
}

export const authApi = {

    request: RegisterRequest,
  ): Promise<ApiResponse<string>> => {
    const response = await axiosClient.post<ApiResponse<string>>(
      '/auth/register',
      request,
    )

    return response.data
  },

  verifyEmail: async (
    request: VerifyEmailRequest,
  ): Promise<ApiResponse<null>> => {
    const response = await axiosClient.get<ApiResponse<null>>(
      '/auth/verify-email',
      { params: request },
    )

    return response.data
  },

  forgotPassword: async (
    email: string,
  ): Promise<ApiResponse<null>> => {
    const response = await axiosClient.post<ApiResponse<null>>(
      '/auth/forgot-password',
      { email },
    )

    return response.data
  },

  login: async (
    request: LoginRequest,
  ): Promise<ApiResponse<LoginResponse>> => {
    const response = await axiosClient.post<
      ApiResponse<LoginResponse> | LoginResponse
    >('/auth/login', request)

    const payload = response.data

    if ('accessToken' in payload) {
      return {
        isSuccess: true,
        statusCode: 200,
        message: '',
        data: payload,
        errors: [],
      }
    }

    return payload
  },

  changePassword: async (
    request: ChangePasswordRequest,
  ): Promise<ApiResponse<null>> => {
    const response = await axiosClient.post<ApiResponse<null> | null>(
      '/auth/change-password',
      request,
    )

    if (!response.data || !('isSuccess' in response.data)) {
      return {
        isSuccess: true,
        statusCode: 200,
        message: '',
        data: null,
        errors: [],
      }
    }

    return response.data
  },

  refresh: async (): Promise<
    ApiResponse<RefreshTokenResponse>
  > => {
    const response =
      await axiosClient.post<
        ApiResponse<RefreshTokenResponse>
      >('/auth/refresh')

    return response.data
  },

  logout: async (): Promise<void> => {
    await axiosClient.post('/auth/logout')
  },
}

import axios, {
  AxiosError,
  AxiosResponse,
  InternalAxiosRequestConfig,
} from 'axios'
import { authStorage } from '../services/authStorage'

const axiosClient = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  withCredentials: true,
  timeout: 10000,
})

type RetryableConfig = InternalAxiosRequestConfig & { _retry?: boolean }

// CÃ¡c endpoint nÃ y tá»± xá»­ lÃ½ 401 cá»§a chÃ­nh chÃºng:
// - login 401 = sai tÃ i khoáº£n/máº­t kháº©u (khÃ´ng Ä‘Æ°á»£c redirect)
// - refresh 401 = háº¿t Ä‘Æ°á»ng sá»‘ng (khÃ´ng Ä‘Æ°á»£c refresh láº¡i -> trÃ¡nh vÃ²ng láº·p)
const AUTH_ENDPOINTS = ['/auth/login', '/auth/refresh']

const isAuthEndpoint = (url?: string): boolean =>
  !!url && AUTH_ENDPOINTS.some((path) => url.includes(path))

/** XÃ³a toÃ n bá»™ dá»¯ liá»‡u Ä‘Äƒng nháº­p vÃ  Ä‘Æ°a ngÆ°á»i dÃ¹ng vá» trang login. */
const forceLogout = () => {
  authStorage.clearAuth()

  if (window.location.pathname !== '/login') {
    window.location.replace('/login')
  }
}

let isRefreshing = false
let failedQueue: Array<{
  resolve: (token: string) => void
  reject: (reason?: unknown) => void
}> = []

const processQueue = (error: unknown, token: string | null = null) => {
  failedQueue.forEach((promise) => {
    if (error || !token) promise.reject(error)
    else promise.resolve(token)
  })
  failedQueue = []
}

// ------------------------------------------------------------
// Request interceptor
// ------------------------------------------------------------
axiosClient.interceptors.request.use(
  (config: InternalAxiosRequestConfig) => {
    const accessToken = authStorage.getAccessToken()

    if (accessToken) {
      config.headers.Authorization = `Bearer ${accessToken}`
    }

    if (config.data instanceof FormData) {
      // KhÃ´ng Ã©p JSON, Ä‘á»ƒ trÃ¬nh duyá»‡t tá»± Ä‘áº·t multipart boundary.
      config.headers.delete('Content-Type')
    } else if (!config.headers.get('Content-Type')) {
      config.headers.set('Content-Type', 'application/json')
    }

    return config
  },
  (error) => Promise.reject(error),
)

// ------------------------------------------------------------
// Response interceptor
// ------------------------------------------------------------
axiosClient.interceptors.response.use(
  (response: AxiosResponse) => response,
  async (error: AxiosError) => {
    const originalRequest = error.config as RetryableConfig | undefined

    if (!originalRequest) return Promise.reject(error)

    // 403: Ä‘Ã£ Ä‘Äƒng nháº­p nhÆ°ng khÃ´ng Ä‘á»§ quyá»n.
    if (error.response?.status === 403) {
      if (window.location.pathname !== '/403') {
        window.location.href = '/403'
      }
      return Promise.reject(error)
    }

    if (error.response?.status !== 401) return Promise.reject(error)

    // 401 tá»« login/refresh: Ä‘á»ƒ nÆ¡i gá»i tá»± xá»­ lÃ½.
    // RiÃªng refresh tháº¥t báº¡i nghÄ©a lÃ  phiÃªn Ä‘Ã£ cháº¿t háº³n -> Ä‘Äƒng xuáº¥t.
    if (isAuthEndpoint(originalRequest.url)) {
      if (originalRequest.url?.includes('/auth/refresh')) forceLogout()
      return Promise.reject(error)
    }

    // ÄÃ£ thá»­ refresh + retry mÃ  váº«n 401 (vÃ­ dá»¥ session bá»‹ revoke): Ä‘Äƒng xuáº¥t.
    if (originalRequest._retry) {
      forceLogout()
      return Promise.reject(error)
    }

    originalRequest._retry = true

    // Äang cÃ³ má»™t láº§n refresh cháº¡y: xáº¿p hÃ ng chá» token má»›i.
    if (isRefreshing) {
      return new Promise<string>((resolve, reject) => {
        failedQueue.push({ resolve, reject })
      }).then((token) => {
        originalRequest.headers.Authorization = `Bearer ${token}`
        return axiosClient(originalRequest)
      })
    }

    isRefreshing = true

    try {
      const newToken = await authStorage.refreshAccessToken()
      processQueue(null, newToken)

      originalRequest.headers.Authorization = `Bearer ${newToken}`
      return await axiosClient(originalRequest)
    } catch (refreshError) {
      processQueue(refreshError)
      forceLogout()
      return Promise.reject(refreshError)
    } finally {
      isRefreshing = false
    }
  },
)

export default axiosClient

      })
    })

    // Ná»‘i láº¡i sau khi máº¥t máº¡ng: bÃ¹ thÃ´ng bÃ¡o Ä‘Ã£ lá»¡
    connection.onreconnected(() => {
      window.dispatchEvent(new Event(NOTIFICATION_RESYNC_EVENT))

      // KhÃ´ng káº¿t ná»‘i Ä‘Æ°á»£c: polling 60 giÃ¢y trong chuÃ´ng váº«n bÃ¹ dá»¯ liá»‡u

const ACCESS_TOKEN_KEY = 'accessToken'

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL
const REQUEST_TIMEOUT = 10000

type StorageType = 'local' | 'session'

const getStorage = (): Storage => {
  const storageType = localStorage.getItem(STORAGE_TYPE_KEY)

  return storageType === 'local' ? localStorage : sessionStorage
}

const getStorageByType = (storageType: StorageType): Storage => {
  return storageType === 'local' ? localStorage : sessionStorage

const clearStorage = (storage: Storage) => {
  storage.removeItem(ACCESS_TOKEN_KEY)
  storage.removeItem(ACCESS_TOKEN_EXPIRES_AT_KEY)
  storage.removeItem(ANTI_CSRF_TOKEN_KEY)

const parseStringArray = (raw: string | null): string[] => {
  if (!raw) return []

  try {
    const parsed = JSON.parse(raw)
    return Array.isArray(parsed) ? parsed : []
  } catch {
    return []
  }
}

/**
 * Gá»i API cáº§n Ä‘Äƒng nháº­p (logout, logout-all...).
 *
 * DÃ¹ng axios thÆ°á»ng (khÃ´ng qua axiosClient) Ä‘á»ƒ trÃ¡nh import vÃ²ng trÃ²n
 * vÃ  trÃ¡nh interceptor tá»± redirect giá»¯a chá»«ng khi Ä‘ang Ä‘Äƒng xuáº¥t.
 * Náº¿u access token Ä‘Ã£ háº¿t háº¡n (401) thÃ¬ refresh má»™t láº§n rá»“i gá»i láº¡i,
 * Ä‘á»ƒ session phÃ­a server váº«n Ä‘Æ°á»£c thu há»“i Ä‘Ãºng cÃ¡ch.
 */
const postAuthorized = async (
  path: string,
  body: unknown,
): Promise<void> => {
  const send = (token: string) =>
    axios.post(`${API_BASE_URL}${path}`, body, {
      withCredentials: true,
      timeout: REQUEST_TIMEOUT,
      headers: { Authorization: `Bearer ${token}` },
    })

  const token = authStorage.getAccessToken()

  // KhÃ´ng cÃ³ token thÃ¬ khÃ´ng cÃ²n gÃ¬ Ä‘á»ƒ thu há»“i phÃ­a server.
  if (!token) return

  try {
    await send(token)
  } catch (error) {
    if (axios.isAxiosError(error) && error.response?.status === 401) {
      const newToken = await authStorage.refreshAccessToken()
      await send(newToken)
      return
    }

    throw error
  }
}

export const authStorage = {
  /**
   * LÆ°u thÃ´ng tin authentication.
   *
   * Refresh Token KHÃ”NG Ä‘Æ°á»£c lÆ°u á»Ÿ Ä‘Ã¢y.
   * Refresh Token Ä‘Æ°á»£c Backend lÆ°u trong HttpOnly Cookie.
   */
  saveAuth: (
    accessToken: string,
    accessTokenExpiresAt: string,
    antiCsrfToken: string,
    roles: string[],
    permissions: string[],
    mustChangePassword: boolean,
    remember: boolean,
  ) => {
    const storageType: StorageType = remember ? 'local' : 'session'
    const storage = getStorageByType(storageType)

    // XÃ³a authentication cÅ© á»Ÿ cáº£ hai storage
    clearStorage(localStorage)
    clearStorage(sessionStorage)

    localStorage.removeItem(STORAGE_TYPE_KEY)
    sessionStorage.removeItem(STORAGE_TYPE_KEY)

    // Ghi nhá»› nÆ¡i lÆ°u auth
    localStorage.setItem(STORAGE_TYPE_KEY, storageType)

    storage.setItem(ACCESS_TOKEN_KEY, accessToken)
    storage.setItem(ACCESS_TOKEN_EXPIRES_AT_KEY, accessTokenExpiresAt)
    storage.setItem(ANTI_CSRF_TOKEN_KEY, antiCsrfToken)
    storage.setItem(ROLES_KEY, JSON.stringify(roles))
    storage.setItem(PERMISSIONS_KEY, JSON.stringify(permissions))
    storage.setItem(MUST_CHANGE_PASSWORD_KEY, String(mustChangePassword))
  },

  /**
   * Cáº­p nháº­t Access Token sau khi refresh.
   * Refresh Token do HttpOnly Cookie quáº£n lÃ½, khÃ´ng Ä‘á»¥ng tá»›i á»Ÿ Ä‘Ã¢y.
   */
  updateAccessToken: (accessToken: string, accessTokenExpiresAt: string) => {
    const storage = getStorage()

    storage.setItem(ACCESS_TOKEN_KEY, accessToken)
    storage.setItem(ACCESS_TOKEN_EXPIRES_AT_KEY, accessTokenExpiresAt)
  },

  getAccessToken: (): string | null => getStorage().getItem(ACCESS_TOKEN_KEY),

  getAccessTokenExpiresAt: (): string | null =>
    getStorage().getItem(ACCESS_TOKEN_EXPIRES_AT_KEY),

  getAntiCsrfToken: (): string | null =>
    getStorage().getItem(ANTI_CSRF_TOKEN_KEY),

  getRoles: (): string[] => parseStringArray(getStorage().getItem(ROLES_KEY)),

  getPermissions: (): string[] =>
    parseStringArray(getStorage().getItem(PERMISSIONS_KEY)),

  mustChangePassword: (): boolean =>
    getStorage().getItem(MUST_CHANGE_PASSWORD_KEY) === 'true',

  completePasswordChange: () => {
    localStorage.setItem(MUST_CHANGE_PASSWORD_KEY, 'false')
    sessionStorage.setItem(MUST_CHANGE_PASSWORD_KEY, 'false')
  },

  clearAuth: () => {
    clearStorage(localStorage)
    clearStorage(sessionStorage)

    localStorage.removeItem(STORAGE_TYPE_KEY)
    sessionStorage.removeItem(STORAGE_TYPE_KEY)
  },

  isAuthenticated: (): boolean => !!authStorage.getAccessToken(),

  /**
   * Láº¥y access token má»›i báº±ng refresh token trong HttpOnly cookie.
   *
   * Backend Báº®T BUá»˜C cÃ³ access token cÅ© (cÃ³ thá»ƒ Ä‘Ã£ háº¿t háº¡n) trong body:
   * nÃ³ Ä‘á»c user id vÃ  session id tá»« token Ä‘Ã³ Ä‘á»ƒ Ä‘á»‘i chiáº¿u vá»›i refresh token.
   */
  refreshAccessToken: async (): Promise<string> => {
    try {
      const currentAccessToken = authStorage.getAccessToken()

      if (!currentAccessToken) {
        throw new Error('No access token available to refresh.')
      }

      const response = await axios.post(
        `${API_BASE_URL}/auth/refresh`,
        { accessToken: currentAccessToken },
        {
          withCredentials: true,
          timeout: REQUEST_TIMEOUT,
        },
      )

      const refreshPayload = response.data?.data ?? response.data

      const {
        accessToken,
        accessTokenExpiresAt,
        antiCsrfToken,
        roles,
        permissions,
      } = refreshPayload

      if (!accessToken) {
        throw new Error('Refresh response did not contain an access token.')
      }

      authStorage.updateAccessToken(accessToken, accessTokenExpiresAt)

      // Cáº­p nháº­t cÃ¡c giÃ¡ trá»‹ phá»¥ náº¿u backend cÃ³ tráº£ vá»
      const storage = getStorage()

      if (antiCsrfToken) {
        storage.setItem(ANTI_CSRF_TOKEN_KEY, antiCsrfToken)
      }

      if (roles) {
        storage.setItem(ROLES_KEY, JSON.stringify(roles))
      }

      if (permissions) {
        storage.setItem(PERMISSIONS_KEY, JSON.stringify(permissions))
      }

      return accessToken
    } catch (error) {
      // Refresh tháº¥t báº¡i = phiÃªn khÃ´ng cÃ²n há»£p lá»‡
      authStorage.clearAuth()
      throw error
    }
  },

  /**
   * ÄÄƒng xuáº¥t thiáº¿t bá»‹ hiá»‡n táº¡i.
   * LuÃ´n dá»n dá»¯ liá»‡u local dÃ¹ API thÃ nh cÃ´ng hay tháº¥t báº¡i.
   * NÆ¡i gá»i chá»‹u trÃ¡ch nhiá»‡m chuyá»ƒn hÆ°á»›ng vá» /login.
   */
  logout: async (reason?: string): Promise<void> => {
    try {
      await postAuthorized('/auth/logout', { reason })
    } catch (error) {
      console.error('Logout failed:', error)
    } finally {

    }
  },

  /**

  logoutAll: async (reason?: string): Promise<void> => {

const getErrorMessage = (error: unknown): string => {
  if (typeof error !== 'object' || error === null) {
    return 'ÄÃ£ xáº£y ra lá»—i. Vui lÃ²ng thá»­ láº¡i.'
  }

  const err = error as {
    message?: string
    response?: {
      status?: number
      data?: unknown
    }
  }

  const data = err.response?.data

  if (typeof data === 'string' && data.trim()) {
    return data
  }

  if (typeof data === 'object' && data !== null) {
    const body = data as {
      code?: unknown
      description?: unknown
      message?: unknown
      title?: unknown
      detail?: unknown
      errors?: unknown
    }

    // Æ¯u tiÃªn thÃ´ng bÃ¡o nghiá»‡p vá»¥ do backend tráº£ vá».
    if (
      typeof body.description === 'string' &&
      body.description.trim()
    ) {
      return body.description
    }

    if (
      typeof body.message === 'string' &&
      body.message.trim()
    ) {
      return body.message
    }

    if (
      typeof body.detail === 'string' &&
      body.detail.trim()
    ) {
      return body.detail
    }

    if (
      typeof body.title === 'string' &&
      body.title.trim()
    ) {
      return body.title
    }

    if (body.errors && typeof body.errors === 'object') {
      const messages = Array.isArray(body.errors)
        ? body.errors.filter(
            (item): item is string =>
              typeof item === 'string' && item.trim().length > 0,
          )
        : Object.values(
            body.errors as Record<string, unknown>,
          ).flatMap((value) =>
            Array.isArray(value)
              ? value.filter(
                  (item): item is string =>
                    typeof item === 'string' &&
                    item.trim().length > 0,
                )
              : typeof value === 'string' && value.trim()
                ? [value]
                : [],
          )

      if (messages.length > 0) {
        return messages.join('\n')
      }
    }
  }

  if (err.response?.status === 400) {
    return 'Dá»¯ liá»‡u khÃ´ng há»£p lá»‡. Vui lÃ²ng kiá»ƒm tra láº¡i thÃ´ng tin.'
  }

  if (err.response?.status === 401) {
    return 'PhiÃªn Ä‘Äƒng nháº­p Ä‘Ã£ háº¿t háº¡n. Vui lÃ²ng Ä‘Äƒng nháº­p láº¡i.'
  }

  if (err.response?.status === 403) {
    return 'Báº¡n khÃ´ng cÃ³ quyá»n thá»±c hiá»‡n thao tÃ¡c nÃ y.'
  }

  return err.message || 'ÄÃ£ xáº£y ra lá»—i. Vui lÃ²ng thá»­ láº¡i.'
}

const tone = (
  c: AudioContext,
  freq: number,
  start: number,
  duration: number,
  volume = 0.13
): void => {
  const osc = c.createOscillator()
  const gain = c.createGain()

  osc.type = 'sine'

  osc.frequency.setValueAtTime(freq, start)

  // Ã‚m thanh tÄƒng nhanh rá»“i giáº£m dáº§n tá»± nhiÃªn
  gain.gain.setValueAtTime(0.0001, start)

  gain.gain.exponentialRampToValueAtTime(
    volume,
    start + 0.025
  )

  gain.gain.exponentialRampToValueAtTime(
    0.0001,
    start + duration
  )

  osc.connect(gain)
  gain.connect(c.destination)

  osc.start(start)
  osc.stop(start + duration + 0.02)

  // Giáº£i phÃ³ng tÃ i nguyÃªn sau khi phÃ¡t
  osc.onended = () => {
    osc.disconnect()
    gain.disconnect()

import { useEffect, useState } from 'react'
import { getErrorMessage } from '../utils/logHelpers'


/** Gá»i hÃ m async má»—i khi `deps` Ä‘á»•i; tá»± bá» qua káº¿t quáº£ cÅ© (trÃ¡nh race condition). */
export function useAsync<T>(fn: () => Promise<T>, deps: unknown[], enabled = true) {
  const [data, setData] = useState<T | null>(null)
  const [loading, setLoading] = useState(enabled)
  const [error, setError] = useState<string | null>(null)
  const [tick, setTick] = useState(0)

  useEffect(() => {
    if (!enabled) {
      setData(null)
      setLoading(false)
      return
    }
    let cancelled = false
    setLoading(true)
    setError(null)

    fn()
      .then((result) => !cancelled && setData(result))
      .catch((err) => !cancelled && setError(getErrorMessage(err)))
      .finally(() => !cancelled && setLoading(false))

    return () => {
      cancelled = true
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [...deps, enabled, tick])

  return { data, loading, error, reload: () => setTick((t) => t + 1) }
}

/** `draft`: giÃ¡ trá»‹ Ä‘ang nháº­p. `applied`: chá»‰ cáº­p nháº­t khi báº¥m "Lá»c" -> dÃ¹ng Ä‘á»ƒ gá»i API. */
export function useFilters<T extends object>(initial: T) {
  const [draft, setDraft] = useState<T>(initial)
  const [applied, setApplied] = useState<T>(initial)

  const patch = (p: Partial<T>) => setDraft((d) => ({ ...d, ...p }))
  const apply = () => setApplied(draft)
  const reset = () => {
    setDraft(initial)
    setApplied(initial)
  }

  return { draft, applied, patch, apply, reset }
}

export const matchesTaskFilter = (
  task: KanbanTask,
  search: string,
  priority: string,
) => {
  const matchesSearch =
    !search ||
    `${task.title} ${task.description || ''}`
      .toLowerCase()
      .includes(search.toLowerCase())

  const priorityMap: Record<string, string> = {
    '1': 'Low',
    '2': 'Medium',
    '3': 'High',
  }

  const matchesPriority =
    !priority || task.priority === priorityMap[priority]

  return matchesSearch && matchesPriority
}

  totalPages: number

}

export const NotificationPagination: React.FC<NotificationPaginationProps> = ({
  page,
  totalPages,
  onPageChange,
}) => {
  if (totalPages <= 1) return null

  return (
    <div className="d-flex justify-content-between align-items-center flex-wrap gap-3 mt-4 pt-3 border-top">
      <div className="text-muted small">
        Trang <strong>{page}</strong> trÃªn tá»•ng <strong>{totalPages}</strong> trang
      </div>

      <CPagination className="mb-0" size="sm">
        <CPaginationItem
          disabled={page <= 1}
          onClick={() => onPageChange(Math.max(1, page - 1))}
          style={{ cursor: page <= 1 ? 'not-allowed' : 'pointer' }}
        >
          &laquo; TrÆ°á»›c
        </CPaginationItem>

        {Array.from({ length: totalPages }, (_, i) => i + 1)
          .filter((p) => p === 1 || p === totalPages || Math.abs(p - page) <= 2)
          .map((p, idx, arr) => {
            const prevP = arr[idx - 1]
            const showEllipsis = prevP && p - prevP > 1

            return (
              <React.Fragment key={p}>
                {showEllipsis && <CPaginationItem disabled>...</CPaginationItem>}
                <CPaginationItem
                  active={p === page}
                  onClick={() => onPageChange(p)}
                  style={{ cursor: 'pointer' }}
                >
                  {p}
                </CPaginationItem>
              </React.Fragment>
            )
          })}

        <CPaginationItem
          disabled={page >= totalPages}
          onClick={() => onPageChange(Math.min(totalPages, page + 1))}
          style={{ cursor: page >= totalPages ? 'not-allowed' : 'pointer' }}
        >
          Sau &raquo;
        </CPaginationItem>

  isSoundEnabled,

export function useNotifications() {
  const navigate = useNavigate()

  // Tab & Core State
  const [tab, setTab] = useState<NotificationKind>('general')
  const [items, setItems] = useState<NotificationDto[]>([])
  const [total, setTotal] = useState(0)
  const [page, setPage] = useState(1)
  const [pageSize, setPageSize] = useState(20)
  const [loading, setLoading] = useState(false)
  const [unread, setUnread] = useState<UnreadCount>({ general: 0, mention: 0 })

  // Filters
  const [search, setSearch] = useState('')
  const [statusFilter, setStatusFilter] = useState<'all' | 'unread' | 'read'>('all')
  const [typeFilter, setTypeFilter] = useState<string>('all')

  // Sound & Action Loading
  const [soundOn, setSoundOn] = useState<boolean>(isSoundEnabled())
  const [actionLoading, setActionLoading] = useState<string | null>(null)

  const seenIds = useRef<Set<string>>(new Set())

  // Unlock audio on interaction
  useEffect(() => {
    const unlock = () => unlockNotificationSound()
    window.addEventListener('pointerdown', unlock, { once: true })
    window.addEventListener('keydown', unlock, { once: true })
    return () => {
      window.removeEventListener('pointerdown', unlock)
      window.removeEventListener('keydown', unlock)
    }
  }, [])

  // Load unread count
  const loadUnread = useCallback(async () => {
    try {
      const data = await notificationService.getUnreadCount()
      setUnread(data)
    } catch (err) {
      console.error('Failed to load unread count:', err)
    }
  }, [])

  // Load notification items
  const loadNotifications = useCallback(
    async (kind: NotificationKind, p = page, size = pageSize) => {
      setLoading(true)
      try {
        const data = await notificationService.getList(kind, p, size)
        data.items.forEach((item) => seenIds.current.add(item.id))
        setItems(data.items)
        setTotal(data.total)
      } catch (err: any) {
        toast.error('KhÃ´ng thá»ƒ táº£i danh sÃ¡ch thÃ´ng bÃ¡o. Vui lÃ²ng thá»­ láº¡i.')
      } finally {
        setLoading(false)
      }
    },
    [page, pageSize],
  )

  // Fetch when tab, page, or pageSize changes
  useEffect(() => {
    void loadNotifications(tab, page, pageSize)
  }, [tab, page, pageSize, loadNotifications])

  // Initial unread fetch
  useEffect(() => {
    void loadUnread()
  }, [loadUnread])

  // Realtime SignalR listener & reconnect sync
  useEffect(() => {
    const handleReceived = (e: Event) => {
      const n = (e as CustomEvent<NotificationDto>).detail
      if (!n?.id || seenIds.current.has(n.id)) return
      seenIds.current.add(n.id)

      playNotificationSound()

      const itemKind = kindOf(n)
      setUnread((prev) => ({
        ...prev,
        [itemKind]: (prev[itemKind] || 0) + 1,
      }))

      // If matches current tab, update list
      if (itemKind === tab) {
        setTotal((prev) => prev + 1)
        if (page === 1) {
          setItems((prev) => [{ ...n, isRead: false }, ...prev])
        }
      }
    }

    const handleResync = () => {
      void loadUnread()
      void loadNotifications(tab, page, pageSize)
    }

    window.addEventListener(NOTIFICATION_RECEIVED_EVENT, handleReceived)
    window.addEventListener(NOTIFICATION_RESYNC_EVENT, handleResync)
    return () => {
      window.removeEventListener(NOTIFICATION_RECEIVED_EVENT, handleReceived)
      window.removeEventListener(NOTIFICATION_RESYNC_EVENT, handleResync)
    }
  }, [tab, page, pageSize, loadNotifications, loadUnread])

  // Change tab handler
  const handleTabChange = (nextTab: NotificationKind) => {
    if (tab === nextTab) return
    setTab(nextTab)
    setPage(1)
    setSearch('')
    setStatusFilter('all')
    setTypeFilter('all')
  }

  // Handle clicking notification item (marks read & navigates)
  const handleItemClick = async (n: NotificationDto) => {
    if (!n.isRead) {
      // Optimistic update
      setItems((prev) => prev.map((x) => (x.id === n.id ? { ...x, isRead: true } : x)))
      const kind = kindOf(n)
      setUnread((prev) => ({ ...prev, [kind]: Math.max(0, prev[kind] - 1) }))

      try {
        await notificationService.markAsRead(n.id)
      } catch (err) {
        console.error('Failed to mark read:', err)
        void loadUnread()
      }
    }

    // Contextual navigation
    if (n.taskId) {
      navigate(`/tasks/${n.taskId}`)
    } else if (n.conversationId) {
      const query = new URLSearchParams({ conversationId: n.conversationId })
      if (n.messageId) query.set('messageId', n.messageId)
      navigate(`/messages?${query.toString()}`)
    }
  }

  // Handle mark single as read button click
  const handleMarkSingle = async (e: React.MouseEvent, n: NotificationDto) => {
    e.stopPropagation()
    if (n.isRead || actionLoading === n.id) return

    setActionLoading(n.id)
    const kind = kindOf(n)
    setItems((prev) => prev.map((x) => (x.id === n.id ? { ...x, isRead: true } : x)))
    setUnread((prev) => ({ ...prev, [kind]: Math.max(0, prev[kind] - 1) }))

    try {
      await notificationService.markAsRead(n.id)
      toast.success('ÄÃ£ Ä‘Ã¡nh dáº¥u Ä‘Ã£ Ä‘á»c')
    } catch (err) {
      toast.error('CÃ³ lá»—i xáº£y ra khi cáº­p nháº­t tráº¡ng thÃ¡i')
      void loadUnread()
    } finally {
      setActionLoading(null)
    }
  }

  // Handle mark all as read
  const handleMarkAll = async () => {
    if (unread[tab] === 0 || actionLoading === 'all') return

    setActionLoading('all')
    setItems((prev) => prev.map((x) => ({ ...x, isRead: true })))
    setUnread((prev) => ({ ...prev, [tab]: 0 }))

    try {
      await notificationService.markAllAsRead(tab)
      toast.success(`ÄÃ£ Ä‘Ã¡nh dáº¥u Ä‘Ã£ Ä‘á»c toÃ n bá»™ má»¥c ${tab === 'general' ? 'thÃ´ng bÃ¡o chung' : 'nháº¯c tÃªn'}`)
    } catch (err) {
      toast.error('CÃ³ lá»—i xáº£y ra khi cáº­p nháº­t táº¥t cáº£')
      void loadUnread()
      void loadNotifications(tab, page, pageSize)
    } finally {
      setActionLoading(null)
    }
  }

  // Sound toggle
  const toggleSound = () => {
    const next = !soundOn
    setSoundOn(next)
    setSoundEnabled(next)
    if (next) {
      unlockNotificationSound()
      playNotificationSound()
      toast.success('ÄÃ£ báº­t Ã¢m thanh thÃ´ng bÃ¡o')
    } else {
      toast('ÄÃ£ táº¯t Ã¢m thanh thÃ´ng bÃ¡o', { icon: 'ðŸ”•' })
    }
  }

    setSearch('')
    setStatusFilter('all')
    setTypeFilter('all')

        const matchTitle = (item.title || '').toLowerCase().includes(query)

          !statusFilter ||
          (statusFilter === 'active'
            ? position.isActive === true
            : position.isActive === false)

        return matchesSearch && matchesStatus
      })
  }, [positions, search, statusFilter])

  return {
    search,
    setSearch,
    statusFilter,
    setStatusFilter,
    filteredPositions,
  }
}

  const [statistics, setStatistics] = useState<ProjectStatistics | null>(null)

      setLoading(false)

    load()

  return {
    project,
    progress,
    statistics,
    loading,

import { useCallback, useEffect, useState } from 'react'
import roleService from 'src/services/roleService'
import type { Role, RoleListResponse } from 'src/types/role'
import { ROLE_DEFAULTS } from '../contants/RoleConstants'


export interface UseRoleDataReturn {
  roles: Role[]
  loading: boolean
  error: string | null
  pagination: RoleListResponse | null
  pageNumber: number
  pageSize: number
  search: string
  sortBy: string
  sortDescending: boolean
  totalRoles: number

  // Actions
  setPageNumber: (page: number) => void
  setSearch: (search: string) => void
  setSortBy: (field: string) => void
  setSortDescending: (descending: boolean) => void
  handleSort: (field: string) => void
  handleSearchChange: (value: string) => void
  loadRoles: () => Promise<void>
  clearError: () => void
}

/**
 * Hook Ä‘á»ƒ quáº£n lÃ½ data loading, pagination, search, sort
 *
 * FIX APPLIED:
 * - Added explicit <boolean>, <number>, <string> generics to useState
 * - This prevents TypeScript from inferring literal types (false, 1, "DisplayName")
 * - Now allows type assignments in handlers properly
 */
export const useRoleData = (): UseRoleDataReturn => {
  // =========================================================
  // STATE
  // =========================================================

  // âœ… Explicit generic types prevent literal type inference
  const [roles, setRoles] = useState<Role[]>([])
  const [loading, setLoading] = useState<boolean>(false)
  const [error, setError] = useState<string | null>(null)
  const [pagination, setPagination] = useState<RoleListResponse | null>(null)

  const [pageNumber, setPageNumber] = useState<number>(ROLE_DEFAULTS.PAGE_NUMBER)
  const [pageSize] = useState<number>(ROLE_DEFAULTS.PAGE_SIZE)
  const [search, setSearch] = useState<string>('')
  const [sortBy, setSortBy] = useState<string>(ROLE_DEFAULTS.SORT_BY)
  const [sortDescending, setSortDescending] = useState<boolean>(ROLE_DEFAULTS.SORT_DESCENDING)

  // =========================================================
  // LOAD DATA
  // =========================================================

  const loadRoles = useCallback(async () => {
    try {
      setLoading(true)
      setError(null)

      const result = await roleService.getAll({
        pageNumber,
        pageSize,
        search: search.trim() || undefined,
        sortBy,
        sortDescending,
      })

      setRoles(result.items)
      setPagination(result)
    } catch (err: any) {
      console.error('Failed to load roles:', err)

      const message =
        err?.response?.data?.message ||
        'KhÃ´ng thá»ƒ táº£i danh sÃ¡ch vai trÃ².'

      setError(message)
      setRoles([])
      setPagination(null)
    } finally {
      setLoading(false)
    }
  }, [pageNumber, pageSize, search, sortBy, sortDescending])

  // =========================================================
  // EFFECTS
  // =========================================================

  useEffect(() => {
    loadRoles()
  }, [loadRoles])

  // =========================================================
  // HANDLERS
  // =========================================================

  // âœ… Now setSortDescending accepts boolean (not just literal false)
  const handleSort = (field: string) => {
    if (sortBy === field) {
      setSortDescending((prev) => !prev)
    } else {
      // âœ… Now setSortBy accepts any string (not just literal "DisplayName")
      setSortBy(field)
      // âœ… Now setSortDescending accepts boolean
      setSortDescending(false)
    }
    setPageNumber(ROLE_DEFAULTS.PAGE_NUMBER)
  }

  // âœ… Now setSearch accepts any string
  const handleSearchChange = (value: string) => {
    setSearch(value)
    setPageNumber(ROLE_DEFAULTS.PAGE_NUMBER)
  }

  const clearError = () => {
    setError(null)
  }

  // =========================================================
  // RETURN
  // =========================================================

  return {
    roles,
    loading,
    error,
    pagination,
    pageNumber,
    pageSize,
    search,
    sortBy,
    sortDescending,
    totalRoles: pagination?.totalCount ?? 0,

    setPageNumber,
    setSearch,
    setSortBy,
    setSortDescending,
    handleSort,
    handleSearchChange,
    loadRoles,
    clearError,
  }
}

 */
export const useRoleDelete = (
  onSuccess: () => Promise<void>,
  onNavigatePreviousPage?: () => void
): UseRoleDeleteReturn => {
  // =========================================================
  // STATE
  // =========================================================

  const [showModal, setShowModal] = useState(false)
  const [selectedRole, setSelectedRole] = useState<Role | null>(null)
  const [loading, setLoading] = useState(false)

  // =========================================================
  // HANDLERS
  // =========================================================

  const openModal = (role: Role) => {

    setShowModal(true)
  }

    setShowModal(false)
  }

  const handleDelete = async (): Promise<boolean> => {

      return false
    }

    try {
      setLoading(true)

      return false

  // RETURN
  // =========================================================

  openEditModal: (role: Role) => void

  code: '',
  displayName: '',
  description: '',
}

/**
 * Hook Ä‘á»ƒ quáº£n lÃ½ form state, validation vÃ  submit
 */
export const useRoleForm = (
  onSuccess: () => Promise<void>
): UseRoleFormReturn => {
  // =========================================================
  // STATE
  // =========================================================

  const [showModal, setShowModal] = useState(false)
  const [editingRole, setEditingRole] = useState<Role | null>(null)
  const [form, setForm] = useState<RoleFormState>(INITIAL_FORM_STATE)
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  // =========================================================
  // HANDLERS
  // =========================================================

  const resetForm = () => {
    setForm(INITIAL_FORM_STATE)
    setError(null)
  }

  const openCreateModal = () => {
    setEditingRole(null)
    resetForm()
    setShowModal(true)
  }

  const openEditModal = (role: Role) => {
    setEditingRole(role)
    setForm({
      code: role.code,
      displayName: role.displayName,
      description: role.description,
    })
    setError(null)
    setShowModal(true)
  }

  const closeModal = () => {
    if (loading) return

    setShowModal(false)
    setEditingRole(null)
    resetForm()
  }

  const clearError = () => {
    setError(null)
  }

  // =========================================================
  // VALIDATION
  // =========================================================

  const validateForm = (): boolean => {
    if (!form.code.trim()) {
      setError(ROLE_MESSAGES.CODE_REQUIRED)
      return false
    }

    if (form.code.trim().length > ROLE_VALIDATION.CODE_MAX) {
      setError(ROLE_MESSAGES.CODE_MAX_LENGTH)
      return false
    }

    if (!form.displayName.trim()) {
      setError(ROLE_MESSAGES.NAME_REQUIRED)
      return false
    }

    if (form.displayName.trim().length > ROLE_VALIDATION.NAME_MAX) {
      setError(ROLE_MESSAGES.NAME_MAX_LENGTH)
      return false
    }

    if (!form.description.trim()) {
      setError(ROLE_MESSAGES.DESC_REQUIRED)
      return false
    }

    if (form.description.trim().length > ROLE_VALIDATION.DESC_MAX) {
      setError(ROLE_MESSAGES.DESC_MAX_LENGTH)
      return false
    }

    return true
  }

  // =========================================================
  // SUBMIT
  // =========================================================

  const handleSubmit = async (
    e: React.FormEvent<HTMLFormElement>

    e.preventDefault()

import { useCallback, useEffect, useMemo, useState } from 'react'

interface UseRolePermissionReturn {

  setSearch: (value: string) => void

  togglePermission: (permissionId: string) => void
  toggleModule: (module: string) => void

  isModuleFullySelected: (module: string) => boolean
  isModulePartiallySelected: (module: string) => boolean

  selectAll: () => void
  clearAll: () => void

  refresh: () => Promise<void>
  save: () => Promise<boolean>
}

export const useRolePermission = (
  roleId: string | null,
): UseRolePermissionReturn => {
  const [permissions, setPermissions] = useState<Permission[]>([])

  const [selectedPermissionIds, setSelectedPermissionIds] =
    useState<Set<string>>(new Set())

  const [loading, setLoading] = useState(false)
  const [saving, setSaving] = useState(false)
  const [error, setError] = useState<string | null>(null)

  const [search, setSearch] = useState('')

  /**
   * Load:
   * 1. All permissions
   * 2. Permissions assigned to current role
   */
  const fetchPermissions = useCallback(async () => {
    if (!roleId) {
      setPermissions([])
      setSelectedPermissionIds(new Set())
      return
    }

    try {
      setLoading(true)
      setError(null)

      const [allResponse, roleResponse] = await Promise.all([
        axiosClient.get('/api/v1/permissions', {
          params: {
            pageNumber: 1,
            pageSize: 100,
            sortBy: 'Module',
            sortDescending: false,
          },
        }),

        axiosClient.get(
          `/api/v1/roles/${roleId}/permissions`,
        ),
      ])


      const allData =
        allResponse.data?.data ?? allResponse.data

      /**
       * GET /roles/{roleId}/permissions
       *
       * data:
       * [...]
       */
      const roleData =
        roleResponse.data?.data ?? roleResponse.data

      const allPermissions: Permission[] =
        allData?.items ?? []

      const rolePermissions: Permission[] =
        Array.isArray(roleData)
          ? roleData
          : []

      setPermissions(allPermissions)

      setSelectedPermissionIds(
        new Set(
          rolePermissions.map(
            (permission) => permission.id,
          ),
        ),
      )
    } catch (err: any) {
      const message =
        err?.response?.data?.message ??
        'KhÃ´ng thá»ƒ táº£i danh sÃ¡ch quyá»n.'

      setError(message)
    } finally {
      setLoading(false)
    }
  }, [roleId])

  useEffect(() => {
    fetchPermissions()
  }, [fetchPermissions])

  /**
   * Permissions after searching.
   */
  const filteredPermissions = useMemo(() => {
    const keyword = search.trim().toLowerCase()

    if (!keyword) {
      return permissions
    }

    return permissions.filter((permission) => {
      return (
        permission.code
          .toLowerCase()
          .includes(keyword) ||
        permission.module
          .toLowerCase()
          .includes(keyword) ||
        permission.action
          .toLowerCase()
          .includes(keyword) ||
        permission.displayName
          .toLowerCase()
          .includes(keyword) ||
        permission.description
          ?.toLowerCase()
          .includes(keyword)
      )
    })
  }, [permissions, search])

  /**
   * Toggle one permission.
   */
  const togglePermission = useCallback(
    (permissionId: string) => {
      setSelectedPermissionIds((current) => {
        const next = new Set(current)

        if (next.has(permissionId)) {
          next.delete(permissionId)
        } else {
          next.add(permissionId)
        }

        return next
      })
    },
    [],
  )

  /**
   * Get permissions belonging to one module.
   */
  const getModulePermissions = useCallback(
    (module: string) => {
      return permissions.filter(
        (permission) =>
          permission.module === module,
      )
    },
    [permissions],
  )

  /**
   * Check if all permissions in module
   * are selected.
   */
  const isModuleFullySelected = useCallback(
    (module: string) => {
      const modulePermissions =
        getModulePermissions(module)

      if (modulePermissions.length === 0) {
        return false
      }

      return modulePermissions.every(
        (permission) =>
          selectedPermissionIds.has(
            permission.id,
          ),
      )
    },
    [
      getModulePermissions,
      selectedPermissionIds,
    ],
  )

  /**
   * Check if some but not all permissions

  const isModulePartiallySelected = useCallback(

        ).length

      return (
        selectedCount > 0 &&
        selectedCount < modulePermissions.length
      )
    },
    [
      getModulePermissions,
      selectedPermissionIds,
    ],
  )

  /**
   * Select / unselect all permissions
   * belonging to a module.
   */
  const toggleModule = useCallback(
    (module: string) => {
      const modulePermissions =
        getModulePermissions(module)

      setSelectedPermissionIds((current) => {
        const next = new Set(current)

        const allSelected =
          modulePermissions.every(
            (permission) =>
              next.has(permission.id),
          )

        if (allSelected) {
          modulePermissions.forEach(
            (permission) => {
              next.delete(permission.id)
            },
          )
        } else {
          modulePermissions.forEach(
            (permission) => {
              next.add(permission.id)
            },
          )
        }

        return next
      })
    },
    [getModulePermissions],
  )

  /**
   * Select all permissions.
   */
  const selectAll = useCallback(() => {
    setSelectedPermissionIds(
      new Set(
        permissions.map(
          (permission) => permission.id,
        ),
      ),
    )
  }, [permissions])

  /**
   * Remove all permissions.
   */
  const clearAll = useCallback(() => {
    setSelectedPermissionIds(new Set())
  }, [])

  /**
   * Save selected permissions.
   *
   * NOTE:
   * Backend must provide:
   *
   * PUT /api/v1/roles/{roleId}/permissions
   *
   * {
   *   permissionIds: [...]
   * }
   */
  const save = useCallback(async () => {
    if (!roleId) {
      return false
    }

    try {
      setSaving(true)
      setError(null)

      await axiosClient.put(
        `/api/v1/roles/${roleId}/permissions`,
        {
          permissionIds: Array.from(
            selectedPermissionIds,
          ),
        },
      )

      return true
    } catch (err: any) {
      const message =
        err?.response?.data?.message ??
        'KhÃ´ng thá»ƒ lÆ°u phÃ¢n quyá»n.'

      setError(message)

      return false
    } finally {
      setSaving(false)
    }
  }, [roleId, selectedPermissionIds])

  return {
    permissions: filteredPermissions,
    selectedPermissionIds,
    loading,
    saving,
    error,

    search,
    setSearch,

    togglePermission,
    toggleModule,

    isModuleFullySelected,
    isModulePartiallySelected,

    selectAll,
    clearAll,

    refresh: fetchPermissions,
    save,
  }
}

import { useMemo, useState } from 'react'
import { User } from 'src/types/user'


export const useUserFilters = (
  users: User[],
) => {
  const [search, setSearch] = useState('')
  const [statusFilter, setStatusFilter] =
    useState('all')
  const [roleFilter, setRoleFilter] =
    useState('all')

  const filteredUsers = useMemo(() => {
    const keyword = search
      .trim()
      .toLowerCase()

    return users.filter((user) => {
      const matchesSearch =
        !keyword ||
        user.username
          .toLowerCase()
          .includes(keyword) ||
        user.email
          .toLowerCase()
          .includes(keyword) ||
        user.fullName
          .toLowerCase()
          .includes(keyword)

      const matchesStatus =
        statusFilter === 'all' ||
        String(user.status) === statusFilter

      const matchesRole =
        roleFilter === 'all' ||
        user.roles?.some(
          (role) =>
            role.id === roleFilter,
        )

      return (
        matchesSearch &&
        matchesStatus &&
        matchesRole
      )
    })
  }, [
    users,
    search,
    statusFilter,
    roleFilter,
  ])

  return {
    search,
    setSearch,

    statusFilter,
    setStatusFilter,

    roleFilter,
    setRoleFilter,

    filteredUsers,
  }
}
